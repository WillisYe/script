// 🎯 测量文本宽度 function measureWidth(text, font) { const canvas = document.createElement('canvas'); const ctx = canvas.getContext('2d'); ctx.font = font; return ctx.measureText(text).width; } // 🎯 居中省略主函数 function truncateMiddle(text, font, maxWidth, ellipsis = '...', ratio = 0.5) { const originalWidth = measureWidth(text, font); // 不需要截断 if (originalWidth <= maxWidth) { return { text, width: originalWidth, truncated: false }; } // 二分查找最佳截断点 let left = 0, right = text.length; let best = null; while (left <= right) { const mid = Math.floor((left + right) / 2); const prefixLen = Math.floor(mid * ratio); const suffixLen = mid - prefixLen; const candidate = text.slice(0, prefixLen) + ellipsis + text.slice(-suffixLen || undefined); const width = measureWidth(candidate, font); if (width <= maxWidth) { best = { text: candidate, width, truncated: true }; left = mid + 1; } else { right = mid - 1; } } return best || { text: ellipsis, width: measureWidth(ellipsis, font), truncated: true }; } // 🚀 使用示例 const result = truncateMiddle(longText, '14px monospace', 300); console.log(result.text); // "very-long-...-FileName.tsx"
