💻 核心代码示例

// 1️⃣ 导入 Pretext (通过 esm.sh CDN) import { prepare, layout, prepareWithSegments, layoutWithLines } from 'https://esm.sh/@chenglou/pretext@0.0.3'; // 2️⃣ 基础用法：测量高度 const text = '你的文本内容...'; const font = '16px sans-serif'; const prepared = prepare(text, font); // 预处理（只做一次） const { height, lineCount } = layout(prepared, 320, 24); // 纯计算！ // 3️⃣ Canvas 渲染：获取每行信息 const preparedSeg = prepareWithSegments(text, font); const { lines } = layoutWithLines(preparedSeg, 320, 24); lines.forEach((line, i) => { ctx.fillText(line.text, 0, i * 24); // 手动渲染每一行 }); // 4️⃣ 收缩包裹：计算最小宽度 let maxWidth = 0; walkLineRanges(preparedSeg, 9999, line => { if (line.width > maxWidth) maxWidth = line.width; }); console.log('最小宽度:', Math.ceil(maxWidth)); // 🎯 完美收缩！
